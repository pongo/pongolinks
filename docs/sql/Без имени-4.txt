PRAGMA foreign_keys = ON;

CREATE TABLE bookmarks (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    url TEXT NOT NULL UNIQUE CHECK(url <> ''),
    title TEXT NOT NULL CHECK(title <> ''),
    description TEXT NOT NULL DEFAULT '',
    is_private BOOLEAN NOT NULL DEFAULT 0 CHECK(is_private IN (0, 1)),  -- 0 = public, 1 = private
    created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Индекс для сортировки по дате обновления
CREATE INDEX idx_bookmarks_updated_at ON bookmarks(updated_at DESC);

CREATE TABLE tags (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL CHECK(name <> ''),  -- оригинальное написание (для отображения)
    name_lower TEXT NOT NULL UNIQUE CHECK(name_lower <> '')  -- нормализованное (на уровне приложения для поддержки юникода)
);

CREATE TABLE bookmark_tags (
    bookmark_id INTEGER NOT NULL,
    tag_id INTEGER NOT NULL,
    PRIMARY KEY (bookmark_id, tag_id),
    FOREIGN KEY (bookmark_id) REFERENCES bookmarks(id) ON DELETE CASCADE,
    FOREIGN KEY (tag_id) REFERENCES tags(id) ON DELETE CASCADE
);

-- Для быстрых выборок "все закладки с таким тегом"
CREATE INDEX idx_bookmark_tags_tag_id ON bookmark_tags (tag_id);

CREATE TABLE related_links (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    bookmark_id INTEGER NOT NULL,
    url TEXT NOT NULL CHECK(url <> ''),
    FOREIGN KEY (bookmark_id) REFERENCES bookmarks(id) ON DELETE CASCADE
);

-- Для выборки всех связанных ссылок конкретной закладки
CREATE INDEX idx_related_links_bookmark_id ON related_links(bookmark_id);

-- Быстрый поиск по URL среди связанных ссылок
CREATE INDEX idx_related_links_url ON related_links(url);

-- Триггер для автоматического обновления поля updated_at при изменении записи
CREATE TRIGGER update_bookmarks_updated_at 
AFTER UPDATE ON bookmarks
FOR EACH ROW
WHEN NEW.updated_at = OLD.updated_at
BEGIN
    UPDATE bookmarks SET updated_at = CURRENT_TIMESTAMP WHERE id = OLD.id;
END;

-- FTS5
CREATE VIRTUAL TABLE bookmarks_fts USING fts5(
    title,
    description,
    content='bookmarks',
    content_rowid='id',
    tokenize='unicode61'  -- корректная работа с Unicode / кириллицей
);

-- INSERT
CREATE TRIGGER bookmarks_ai AFTER INSERT ON bookmarks BEGIN
    INSERT INTO bookmarks_fts(rowid, title, description)
    VALUES (new.id, new.title, new.description);
END;

-- UPDATE (сначала удаляем старую запись из индекса)
CREATE TRIGGER bookmarks_bu BEFORE UPDATE OF title, description ON bookmarks BEGIN
    INSERT INTO bookmarks_fts(bookmarks_fts, rowid, title, description)
    VALUES ('delete', old.id, old.title, old.description);
END;

CREATE TRIGGER bookmarks_au AFTER UPDATE OF title, description ON bookmarks BEGIN
    INSERT INTO bookmarks_fts(rowid, title, description)
    VALUES (new.id, new.title, new.description);
END;

-- DELETE
CREATE TRIGGER bookmarks_bd BEFORE DELETE ON bookmarks BEGIN
    INSERT INTO bookmarks_fts(bookmarks_fts, rowid, title, description)
    VALUES ('delete', old.id, old.title, old.description);
END;
